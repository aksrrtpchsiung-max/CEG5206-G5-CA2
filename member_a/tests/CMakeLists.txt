find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(fixture_header "${CMAKE_CURRENT_BINARY_DIR}/fixture.hpp")
set(fixture_json "${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/contract_v0_1.json")
add_custom_command(OUTPUT "${fixture_header}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/fixture.py"
            "${fixture_json}" "${fixture_header}"
    DEPENDS fixture.py "${fixture_json}" VERBATIM)
add_executable(a_correctness main.cpp test_inputs.cpp test_kernels.cpp grid.cpp "${fixture_header}")
target_include_directories(a_correctness PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(a_correctness PRIVATE ceg5206_a)

set(A_B_SOURCE "" CACHE FILEPATH "Optional production B row-range and SIMD source")
set(A_C_SOURCE "" CACHE FILEPATH "Optional production C contract adapter")
set(A_C_TEST_ROW_SOURCE "" CACHE FILEPATH "C's test-only row implementation for native race checks")
if(A_C_TEST_ROW_SOURCE AND (A_B_SOURCE OR NOT A_C_SOURCE))
    message(FATAL_ERROR "The test row source is only for C diagnostics without production B")
endif()
if(A_C_SOURCE AND NOT (A_B_SOURCE OR A_C_TEST_ROW_SOURCE))
    message(FATAL_ERROR "A_C_SOURCE also needs B's row-range implementation")
endif()
if(A_B_SOURCE)
    target_sources(a_correctness PRIVATE "${A_B_SOURCE}")
    target_compile_definitions(a_correctness PRIVATE A_CHECK_B)
endif()
if(A_C_SOURCE)
    find_package(OpenMP REQUIRED)
    target_link_libraries(a_correctness PRIVATE OpenMP::OpenMP_CXX)
    target_sources(a_correctness PRIVATE "${A_C_SOURCE}")
    target_compile_definitions(a_correctness PRIVATE A_CHECK_C)
    if(A_C_TEST_ROW_SOURCE)
        target_sources(a_correctness PRIVATE "${A_C_TEST_ROW_SOURCE}")
    endif()
endif()
if(MSVC)
    target_compile_options(a_correctness PRIVATE /fp:strict)
else()
    target_compile_options(a_correctness PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()
foreach(suite validation generator comparison chunks handoff variants hub)
    add_test(NAME a_${suite} COMMAND a_correctness ${suite})
endforeach()
