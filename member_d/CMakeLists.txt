cmake_minimum_required(VERSION 3.16)
project(ceg5206_member_d LANGUAGES CXX)

add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../member_a" member_a_contracts)
option(D_ENABLE_A "Link A implementations" OFF)
option(D_ENABLE_B "Link B implementation (requires A)" OFF)
option(D_ENABLE_C "Link C implementations (requires A and B)" OFF)
set(D_A_SOURCES "" CACHE STRING "Absolute A source paths, excluding scalar kernel")
set(D_SCALAR_SOURCES "" CACHE STRING "Absolute scalar-only source paths")
set(D_B_SOURCES "" CACHE STRING "Absolute B source paths")
set(D_C_SOURCES "" CACHE STRING "Absolute C source paths")
set(D_EXTRA_LIBRARIES "" CACHE STRING "Extra libraries/targets, e.g. OpenMP::OpenMP_CXX")
option(D_USE_OPENMP "Find and link OpenMP" OFF)

if(D_ENABLE_B AND NOT D_ENABLE_A)
  message(FATAL_ERROR "B execution requires A generation and validation")
endif()
if(D_ENABLE_C AND NOT D_ENABLE_B)
  message(FATAL_ERROR "C execution requires A and B")
endif()
add_executable(benchmark src/benchmark.cpp src/allocation_tracker.cpp)
target_link_libraries(benchmark PRIVATE ceg5206_contracts ${D_EXTRA_LIBRARIES})
target_include_directories(benchmark PRIVATE include)
target_compile_definitions(benchmark PRIVATE D_ALLOCATION_TRACKER)
if(MSVC)
  message(FATAL_ERROR "v0.1 build flags support Clang/GCC; agree MSVC flags before use")
endif()
target_compile_options(benchmark PRIVATE -O3 -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
foreach(member A B C)
  if(D_ENABLE_${member})
    if(NOT D_${member}_SOURCES)
      message(FATAL_ERROR "Provide D_${member}_SOURCES when enabling ${member}")
    endif()
    target_sources(benchmark PRIVATE ${D_${member}_SOURCES})
    target_compile_definitions(benchmark PRIVATE D_ENABLE_${member})
  endif()
endforeach()
if(D_ENABLE_B)
  target_sources(benchmark PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../member_b/bench/edge_major.cpp")
  target_compile_definitions(benchmark PRIVATE D_ENABLE_EDGE_MAJOR)
endif()
if(D_ENABLE_A)
  if(NOT D_SCALAR_SOURCES)
    message(FATAL_ERROR "A scalar kernel must be supplied in its own translation unit")
  endif()
  add_library(d_scalar OBJECT ${D_SCALAR_SOURCES})
  target_link_libraries(d_scalar PRIVATE ceg5206_contracts)
  target_compile_options(d_scalar PRIVATE -O3 -fno-fast-math -ffp-contract=off)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(d_scalar PRIVATE -fno-vectorize -fno-slp-vectorize)
  elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(d_scalar PRIVATE -fno-tree-vectorize)
  else()
    message(FATAL_ERROR "Agree scalar no-vectorization flags for this compiler")
  endif()
  target_sources(benchmark PRIVATE $<TARGET_OBJECTS:d_scalar>)
endif()
if(D_USE_OPENMP)
  find_package(OpenMP REQUIRED)
  target_link_libraries(benchmark PRIVATE OpenMP::OpenMP_CXX)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_measurement tests/test_measurement.cpp)
  target_include_directories(test_measurement PRIVATE include)
  target_link_libraries(test_measurement PRIVATE ceg5206_contracts)
  add_test(NAME measurement COMMAND test_measurement)
endif()

# Optional diagnostic binary: instrumentation never enters production timing.
if(D_ENABLE_C)
  get_target_property(d_sources benchmark SOURCES)
  add_executable(benchmark_profile ${d_sources})
  target_include_directories(benchmark_profile PRIVATE include)
  target_compile_definitions(benchmark_profile PRIVATE D_ENABLE_A D_ENABLE_B D_ENABLE_C CEG5206_D_DIAGNOSTICS D_ALLOCATION_TRACKER D_ENABLE_EDGE_MAJOR)
  target_compile_options(benchmark_profile PRIVATE -O3 -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
  target_link_libraries(benchmark_profile PRIVATE ceg5206_contracts OpenMP::OpenMP_CXX ${D_EXTRA_LIBRARIES})
endif()
if(BUILD_TESTING)
  add_executable(test_allocation tests/test_allocation.cpp src/allocation_tracker.cpp)
  target_compile_features(test_allocation PRIVATE cxx_std_17)
  target_compile_definitions(test_allocation PRIVATE D_ALLOCATION_TRACKER)
  target_include_directories(test_allocation PRIVATE include)
  add_test(NAME allocation COMMAND test_allocation)
endif()
